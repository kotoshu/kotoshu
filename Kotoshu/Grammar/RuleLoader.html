def load_rules
  rules_file = File.join(@rules_path, 'rules.yaml')
  return [] unless File.exist?(rules_file)

  config = YAML.load_file(rules_file)
  return [] unless config && config['rules']

  config['rules'].map { |rule_config| Rule.from_yaml(rule_config) }
end
