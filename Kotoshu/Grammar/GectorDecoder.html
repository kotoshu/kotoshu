Overview

Iterative GECToR-style decoding (Omelianchuk et al. 2020): predict labels, apply confident corrections, re-tag, repeat — multi-edit sentences converge across rounds.

def correct(words)
  tokens = words.dup
  applied = []
  @rounds.times do
    findings = detect_with_confidence(tokens)
    changed = false
    findings.each do |f|
      next if f[:confidence] < threshold_for(f[:label])

      result = apply(f[:label], tokens, f[:index])
      next if result.nil?

      tokens = result
      applied << f
      changed = true
    end
    break unless changed
  end
  { corrected: tokens, applied: applied }
end
