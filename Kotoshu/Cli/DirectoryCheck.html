def initialize(roots, check:, baseline_path: nil, include_globs: [], exclude_globs: [])
  @roots = Array(roots)
  @check = check
  @baseline_path = baseline_path
  @walker = DirectoryWalker.new(include_globs: include_globs, exclude_globs: exclude_globs)
  @outcomes = []
end
