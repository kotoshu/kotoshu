Overview

def from_file(path)
  File.foreach(path, chomp: true) do |line|
    next if line.empty? || line.start_with?("#")

    add_word(line)
  end
  self
end

def from_string(text)
  text.each_line do |line|
    word = line.strip
    next if word.empty? || word.start_with?("#")

    add_word(word)
  end
  self
end
